function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. INPUT FORM BARANG MASUK (Kolom B-F Nilai, Kolom A Rumus ;)
  if (data.action === 'input_fbm') {
    var sheet = ss.getSheetByName('FORM BARANG MASUK');
    if (!sheet) sheet = ss.insertSheet('FORM BARANG MASUK');
    var row = sheet.getLastRow() + 1;
    sheet.getRange(row, 2).setValue(data.tglMasuk);
    sheet.getRange(row, 3).setValue(data.namaBarang);
    sheet.getRange(row, 4).setValue(data.satuan);
    sheet.getRange(row, 5).setValue(data.inQty);
    sheet.getRange(row, 6).setValue(data.divisi);
    
    // Rumus Kolom A dengan pemisah semicolon (;)
    var fbmFormula = '=IF(C' + row + '="";"";("("&F' + row + '&")"&C' + row + '&"("&D' + row + '&")"&TEXT(COUNTA(B$2:B' + row + ');"0000")))';
    try {
      sheet.getRange(row, 1).setFormula(fbmFormula);
    } catch(err) {
      sheet.getRange(row, 1).setFormula('=IF(C' + row + '="","",("("&F' + row + '&")"&C' + row + '&"("&D' + row + '&")"&TEXT(COUNTA(B$2:B' + row + '),"0000")))');
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'success', row: row})).setMimeType(ContentService.MimeType.JSON);
  }

  // 2. INPUT PENGELUARAN SR (Kolom B, D Nilai; Kolom A, C, E Rumus ;)
  if (data.action === 'input_sr') {
    var sheetName = data.sheetName || 'SR1';
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) sheet = ss.insertSheet(sheetName);
    var row = sheet.getLastRow() + 1;
    sheet.getRange(row, 2).setValue(data.kodeBarang);
    sheet.getRange(row, 4).setValue(data.out);
    
    // Rumus Kolom A, C, E dengan pemisah semicolon (;)
    var formA = '=IFNA(XLOOKUP(B' + row + ';INVENTORY!A:A;INVENTORY!B:B;"-";0);"")';
    var formC = '=IFNA(XLOOKUP(B' + row + ';INVENTORY!A:A;INVENTORY!I:I;0;0);"")';
    var formE = '=IF(D' + row + '<>"";"' + sheetName + '";"")';
    
    try {
      sheet.getRange(row, 1).setFormula(formA);
      sheet.getRange(row, 3).setFormula(formC);
      sheet.getRange(row, 5).setFormula(formE);
    } catch(err) {
      sheet.getRange(row, 1).setFormula('=IFNA(XLOOKUP(B' + row + ',INVENTORY!A:A,INVENTORY!B:B,"-",0),"")');
      sheet.getRange(row, 3).setFormula('=IFNA(XLOOKUP(B' + row + ',INVENTORY!A:A,INVENTORY!I:I,0,0),"")');
      sheet.getRange(row, 5).setFormula('=IF(D' + row + '<>"","' + sheetName + '","")');
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'success', row: row})).setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput(JSON.stringify({status: 'ignored'})).setMimeType(ContentService.MimeType.JSON);
}
